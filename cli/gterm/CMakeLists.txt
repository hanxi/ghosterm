cmake_minimum_required(VERSION 3.19)
project(ghosterm_cli C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Independent user-installed artifact, never added to App packaging.
set(GHOSTERM_BUILD_TESTS OFF CACHE BOOL "Build Core tests")
add_subdirectory(../../core core)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-sdk/sdk-version.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../packages/plugin-sdk/sdk-version.json" sdk_identity)
string(JSON sdk_version GET "${sdk_identity}" sdkApiVersion)
string(JSON sdk_protocol GET "${sdk_identity}" bridgeProtocol)
add_executable(gterm main.c stdio.c)
target_compile_definitions(gterm PRIVATE GTERM_SDK_VERSION="${sdk_version}" GTERM_PROTOCOL="${sdk_protocol}")
target_link_libraries(gterm PRIVATE ghosterm_core)
target_compile_options(gterm PRIVATE -Wall -Wextra -Werror)
option(GHOSTERM_GTERM_TESTS "Build explicit CLI process fixtures" OFF)
if(GHOSTERM_GTERM_TESTS)
  add_executable(test_gterm_stdio tests/stdio.c)
  target_link_libraries(test_gterm_stdio PRIVATE ghosterm_core)
  target_compile_options(test_gterm_stdio PRIVATE -Wall -Wextra -Werror)
endif()
