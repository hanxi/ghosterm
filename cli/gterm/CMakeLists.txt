cmake_minimum_required(VERSION 3.19)
project(ghosterm_cli C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Independent user-installed artifact, never added to App packaging.
set(GHOSTERM_BUILD_TESTS OFF CACHE BOOL "Build Core tests")
add_subdirectory(../../core core)
set(sdk_version "1.0.0")
set(sdk_protocol 1)
set(gterm_runtime "lua")
add_executable(gterm main.c stdio.c ipc.c foreground.c catalog.c catalog-store.c catalog-policy.c release-auth.c release-install.c release-store.c distribution.c)
set(gterm_devkit "${CMAKE_CURRENT_BINARY_DIR}/gterm-devkit/scripts")
foreach(tool gterm-plugin plugin_lua_manifest.py plugin_release.py lib/gt-timeout.sh install-gterm.sh)
  configure_file("../../scripts/${tool}" "${gterm_devkit}/${tool}" COPYONLY)
endforeach()
install(TARGETS gterm RUNTIME DESTINATION bin)
install(PROGRAMS "${gterm_devkit}/gterm-plugin" DESTINATION bin/gterm-devkit/scripts)
install(FILES "${gterm_devkit}/plugin_lua_manifest.py" "${gterm_devkit}/plugin_release.py"
  DESTINATION bin/gterm-devkit/scripts)
install(FILES "${gterm_devkit}/install-gterm.sh" DESTINATION bin/gterm-devkit/scripts)
set(GTERM_UPDATE_PUBLIC_KEY "" CACHE STRING "Raw Ed25519 distribution public key (hex)")
if(GTERM_UPDATE_PUBLIC_KEY)
  string(LENGTH "${GTERM_UPDATE_PUBLIC_KEY}" key_length)
  if(NOT GHOSTERM_WITH_RELEASE_SIGNATURE OR NOT key_length EQUAL 64 OR
     NOT GTERM_UPDATE_PUBLIC_KEY MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Distribution updates require release signature support and a 64-digit lowercase public key")
  endif()
  target_compile_definitions(gterm PRIVATE GTERM_UPDATE_KEY="${GTERM_UPDATE_PUBLIC_KEY}")
endif()
install(FILES ../../LICENSE DESTINATION share/licenses/gterm)
if(GHOSTERM_WITH_PLUGIN_LUA)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/core/lua-5.5.1/doc/readme.html"
    DESTINATION share/licenses/gterm RENAME Lua-5.5.1.html)
endif()
install(FILES "${gterm_devkit}/lib/gt-timeout.sh" DESTINATION bin/gterm-devkit/scripts/lib)
if(GHOSTERM_WITH_PLUGIN_LUA AND GHOSTERM_WITH_RELEASE_SIGNATURE)
  find_package(ZLIB REQUIRED)
  target_compile_definitions(gterm PRIVATE GTERM_HAVE_RELEASE=1)
  target_link_libraries(gterm PRIVATE ZLIB::ZLIB)
  if(APPLE)
    find_library(GTERM_CORE_FOUNDATION CoreFoundation REQUIRED)
    target_link_libraries(gterm PRIVATE ${GTERM_CORE_FOUNDATION})
  else()
    option(GTERM_STATIC_ICU "Link static ICU into the distribution binary" OFF)
    if(GTERM_STATIC_ICU)
      find_package(ICU REQUIRED COMPONENTS uc data)
      enable_language(CXX)
      set_property(TARGET gterm PROPERTY LINKER_LANGUAGE CXX)
      target_link_libraries(gterm PRIVATE ICU::uc ICU::data ${CMAKE_DL_LIBS})
    else()
      find_package(ICU REQUIRED COMPONENTS uc)
      target_link_libraries(gterm PRIVATE ICU::uc)
    endif()
  endif()
endif()
target_compile_definitions(gterm PRIVATE GTERM_SDK_VERSION="${sdk_version}" GTERM_PROTOCOL="${sdk_protocol}"
  GTERM_NATIVE_RUNTIME="${gterm_runtime}")
target_link_libraries(gterm PRIVATE ghosterm_core)
target_include_directories(gterm PRIVATE ../../core/src)
target_compile_options(gterm PRIVATE -Wall -Wextra -Werror)
option(GHOSTERM_GTERM_TESTS "Build explicit CLI process fixtures" OFF)
option(GHOSTERM_GTERM_CATALOG_FAULTS "Inject catalog IO failures in owned CLI tests" OFF)
if(GHOSTERM_GTERM_CATALOG_FAULTS)
  if(NOT GHOSTERM_GTERM_TESTS)
    message(FATAL_ERROR "Catalog fault injection requires explicit CLI tests")
  endif()
  target_sources(gterm PRIVATE tests/catalog-faults.c)
  set_source_files_properties(catalog-store.c PROPERTIES COMPILE_DEFINITIONS "fsync=gterm_test_fsync;close=gterm_test_close")
endif()
option(GHOSTERM_GTERM_STORE_FAULTS "Pause owned version deletion after one durable unlink" OFF)
if(GHOSTERM_GTERM_STORE_FAULTS)
  if(NOT GHOSTERM_GTERM_TESTS)
    message(FATAL_ERROR "Store fault injection requires explicit CLI tests")
  endif()
  set_source_files_properties(release-store.c PROPERTIES COMPILE_DEFINITIONS GTERM_STORE_FAULTS=1)
endif()
if(GHOSTERM_GTERM_TESTS)
  add_executable(test_gterm_stdio tests/stdio.c)
  target_link_libraries(test_gterm_stdio PRIVATE ghosterm_core)
  target_compile_options(test_gterm_stdio PRIVATE -Wall -Wextra -Werror)
  if(GHOSTERM_WITH_PLUGIN_LUA)
    foreach(mode live multiplex)
      add_executable(test_gterm_${mode} tests/${mode}.c)
      target_include_directories(test_gterm_${mode} PRIVATE ../../core/src)
      target_compile_definitions(test_gterm_${mode} PRIVATE GT_HAVE_PLUGIN_LUA=1)
      target_link_libraries(test_gterm_${mode} PRIVATE ghosterm_core)
      target_compile_options(test_gterm_${mode} PRIVATE -Wall -Wextra -Werror)
    endforeach()
  endif()
  add_executable(test_gterm_codex tests/codex.c)
  target_link_libraries(test_gterm_codex PRIVATE ghosterm_core)
  target_compile_options(test_gterm_codex PRIVATE -Wall -Wextra -Werror)
  add_executable(test_gterm_codex_peer tests/codex-peer.c)
  target_link_libraries(test_gterm_codex_peer PRIVATE ghosterm_core)
  target_compile_options(test_gterm_codex_peer PRIVATE -Wall -Wextra -Werror)
endif()
